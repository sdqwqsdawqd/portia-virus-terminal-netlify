import { uploadImage } from "@/lib/upload-image";
import { dossierToContent } from "./portia-template-dossier.ts";

export async function saveTemplate(dossier: unknown, file: File) {
  const url = await uploadImage(file);
  await fetch("/api/content", { method: "PUT", body: JSON.stringify({ dossier, url }) });
  sessionStorage.setItem("template", JSON.stringify(dossier));
  return dossierToContent(dossier);
}
